.then(function(gd) {
                        var steps = (gd.layout.sliders && gd.layout.sliders[0] && gd.layout.sliders[0].steps) || [];

                        // hide the built-in Plotly slider so it doesn't show alongside the HTML one
                        Plotly.relayout(gd, {sliders: []});

                        var slider = document.getElementById("strain-slider");
                        var label = document.getElementById("strain-label");
                        slider.max = Math.max(steps.length - 1, 0);

                        function applyStep(k) {
                            var step = steps[k];
                            if (!step) return;
                            Plotly.restyle(gd, step.args[0]);
                            label.innerText = step.label;
                        }

                        slider.addEventListener("input", function(e) {
                            applyStep(parseInt(e.target.value, 10));
                        });

                        if (steps.length) applyStep(0);
                    })
